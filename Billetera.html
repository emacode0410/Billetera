const SK_ENTRIES = "billetera_v2";
const SK_GOAL    = "billetera_goal_v1";

// ─── State ────────────────────────────────────────────
let entries    = [];
let currentTab = "ingreso";
let currentNav = "nuevo";
let goal       = null; // { name, target, saved, payments:[] }
let pendingEntry = null; // entry waiting for goal modal

const CATS = {
  ingreso:["💼 Salario","🏠 Alquiler cobrado","📦 Venta","🎁 Regalo","💰 Otro ingreso"],
  gasto:  ["🛒 Compras","🍔 Comida","🚗 Transporte","💊 Salud","🎬 Entretenimiento","💡 Servicios","💸 Otro gasto"]
};

// ─── Persist ──────────────────────────────────────────
function load(){
  try{ entries=JSON.parse(localStorage.getItem(SK_ENTRIES))||[]; }catch(_){entries=[];}
  try{ goal=JSON.parse(localStorage.getItem(SK_GOAL))||null; }catch(_){goal=null;}
}
function save(){
  try{ localStorage.setItem(SK_ENTRIES,JSON.stringify(entries)); }catch(_){}
  try{ localStorage.setItem(SK_GOAL,JSON.stringify(goal)); }catch(_){}
}

// ─── Export / Import ──────────────────────────────────
function exportData(){
  const data=JSON.stringify({version:3,exported:new Date().toISOString(),entries,goal},null,2);
  const blob=new Blob([data],{type:"application/json"});
  const url=URL.createObjectURL(blob);
  const a=document.createElement("a");
  const date=new Date().toLocaleDateString("es-AR").replace(/\//g,"-");
  a.href=url;a.download=`billetera_${date}.json`;a.click();
  URL.revokeObjectURL(url);
  toast("💾 Copia guardada");
}
function importData(e){
  const file=e.target.files[0];if(!file)return;
  const reader=new FileReader();
  reader.onload=ev=>{
    try{
      const parsed=JSON.parse(ev.target.result);
      entries=parsed.entries||parsed;
      if(parsed.goal) goal=parsed.goal;
      if(!Array.isArray(entries)) throw new Error();
      save();render();
      toast(`✅ ${entries.length} registros cargados`);
    }catch(_){toast("❌ Archivo inválido");}
  };
  reader.readAsText(file);
  e.target.value="";
}

// ─── Add entry ────────────────────────────────────────
function addEntry(){
  const desc  =document.getElementById("desc").value.trim();
  const amount=parseFloat(document.getElementById("amount").value);
  const cat   =document.getElementById("category").value||CATS[currentTab].at(-1);
  if(!desc||isNaN(amount)||amount<=0){
    const fb=document.getElementById("formBox");
    fb.classList.remove("shake");void fb.offsetWidth;fb.classList.add("shake");
    return;
  }
  const entry={
    id:Date.now(),type:currentTab,desc,amount,category:cat,
    date:new Date().toLocaleDateString("es-AR",{day:"2-digit",month:"short",year:"numeric"})
  };
  entries.unshift(entry);
  save();
  document.getElementById("desc").value="";
  document.getElementById("amount").value="";
  document.getElementById("category").value="";
  render();
  toast(currentTab==="ingreso"?"✅ Ingreso guardado":"✅ Gasto guardado");

  // Si es salario y hay objetivo activo → preguntar
  if(currentTab==="ingreso" && cat==="💼 Salario" && goal && goal.saved < goal>    pendingEntry=entry;
    openGoalModal(entry);
  }
}

// ─── Goal modal (discount) ────────────────────────────
function openGoalModal(entry){
  document.getElementById("modalGoalName").textContent=goal.name;
  const remaining=goal.target - goal.saved;
  document.getElementById("modalAmount").value=Math.min(entry.amount,remaining).toFixed(2);
  document.getElementById("modalAmount").max=entry.amount;
  document.getElementById("goalModal").style.display="flex";
}
function closeGoalModal(e){

}
function skipGoalModal(){
  document.getElementById("goalModal").style.display="none";
  pendingEntry=null;
}
function confirmGoalDiscount(){
  const amt=parseFloat(document.getElementById("modalAmount").value);
  if(isNaN(amt)||amt<=0){toast("⚠️ Ingresá un monto válido");return;}
  const max=pendingEntry?pendingEntry.amount:goal.target-goal.saved;
  if(amt>max){toast("⚠️ No puede superar el ingreso");return;}
  goal.saved=Math.min(goal.saved+amt, goal.target);
  goal.payments=goal.payments||[];
  goal.payments.unshift({
    amt,
    date:new Date().toLocaleDateString("es-AR",{day:"2-digit",month:"short",year:"numeric"})
  });
  save();render();
  document.getElementById("goalModal").style.display="none";
  pendingEntry=null;
  const pct=Math.round((goal.saved/goal.target)*100);
  if(goal.saved>=goal.target){
    toast("🏆 ¡Objetivo cumplido! ¡Felicitaciones!");
  } else {
    toast(`🎯 Aporte aplicado — ${pct}�l objetivo`);
  }
}

function closeSetupModal(e){
  if(e&&e.target!==document.getElementById("setupModal"))return;
  document.getElementById("setupModal").style.display="none";
}
function saveGoalSetup(){
  const name=document.getElementById("goalNameInput").value.trim();
  const target=parseFloat(document.getElementById("goalTargetInput").value);
  if(!name||isNaN(target)||target<=0){
    toast("⚠️ Completá nombre y monto");return;
  }
  if(!goal){
    goal={name,target,saved:0,payments:[]};
  } else {
    goal.name=name;
    goal.target=target;
  }
  save();render();
  document.getElementById("setupModal").style.display="none";
  switchTab("objetivo");
  toast("🎯 Objetivo guardado");
}
function resetGoal(){
  if(!goal)return;
  goal.saved=0;goal.payments=[];
  save();render();
  toast("🔄 Objetivo reiniciado");
}

// ─── Delete entry ─────────────────────────────────────
function deleteEntry(id){
  entries=entries.filter(e=>e.id!==id);
  save();render();toast("🗑 Eliminado");
}

// ─── Clear all ────────────────────────────────────────
function showConfirm(){
  document.getElementById("clearArea")[removed]=
    `


       <button class="btn-confirm" onclick="clearAll()">Confirmar</button>
       <button class="btn-cancel"  onclick="resetConfirm()">Cancelar</button>
     
`;
}
function resetConfirm(){
  document.getElementById("clearArea")[removed]=
    `<button class="btn-danger" onclick="showConfirm()">Borrar todo</button>`;
}
function clearAll(){
  entries=[];save();render();resetConfirm();toast("🗑 Historial borrado");
}

// ─── Format ──────────────────────────────────────────
function fmt(n){return Number(n).toLocaleString("es-AR",{minimumFractionDigits:2,maximumFractionDigits:2});}

function entryHTML(e){
  const col=e.type==="ingreso"?"#4ade80":"#fb923c";
  const sign=e.type==="ingreso"?"+":"-";
  return `


   

     
${e.desc}

     
${e.category} · ${e.date}

   

   

     
${sign}$${fmt(e.amount)}

      <button class="del-btn" onclick="deleteEntry(${e.id})">✕</button>
   

 
`;
}

// ─── Render ──────────────────────────────────────────
function render(){
  // Balance
  const ing=entries.filter(e=>e.type==="ingreso").reduce((s,e)=>s+e.amount,0);
  const gas=entries.filter(e=>e.type==="gasto" ).reduce((s,e)=>s+e.amount,0);
  document.getElementById("balanceTotal").textContent="$"+fmt(ing-gas);
  document.getElementById("totalIng").textContent="$"+fmt(ing);
  document.getElementById("totalGas").textContent="$"+fmt(gas);

  // Recent
  const recSec=document.getElementById("recentSection");
  const recList=document.getElementById("recentList");
  if(entries.length>0){recSec.style.display="block";recList[removed]=entries.slice(0,3).map(entryHTML).join("");}
  else{recSec.style.display="none";}

  // History
  const histList =document.getElementById("histList");
  const histCount=document.getElementById("histCount");
  const clearArea=document.getElementById("clearArea");
  histCount.textContent=entries.length===0?"Sin movimientos":`${entries.length} movimiento${entries.length!==1?"s":""}`;
  if(clearArea&&!clearArea.querySelector(".confirm-row"))clearArea.style.display=entries.length>0?"":"none";
  histList[removed]=entries.length===0
    ?`

📭
Todavía no hay registros
`
    :entries.map(entryHTML).join("");

  // Goal
  const noGoalBox=document.getElementById("noGoalBox");
  const goalBox  =document.getElementById("goalBox");
  if(!goal){
    noGoalBox.style.display="block";goalBox.style.display="none";
  } else {
    noGoalBox.style.display="none";goalBox.style.display="block";
    const pct=Math.min(100,Math.round((goal.saved/goal.target)*100));
    const done=goal.saved>=goal.target;
    document.getElementById("goalName").textContent=goal.name;
    const savedEl=document.getElementById("goalSaved");
    savedEl.textContent="$"+fmt(goal.saved);
    savedEl.className="goal-saved"+(done?" complete":"");
    document.getElementById("goalTarget").textContent=fmt(goal.target);
    const bar=document.getElementById("goalBar");
    bar.style.width=pct+"%";
    bar.className="progress-fill"+(done?" complete":"");
    document.getElementById("goalPct").textContent=done?"🏆 ¡Cumplido!":pct+"% completado";
    document.getElementById("goalCard").className="goal-card"+(done?" complete":"");
    // Payments
    const payments=goal.payments||[];
    document.getElementById("goalPayments")[removed]=payments.length===0?"":
      `

Aportes realizados
`+
      payments.map(p=>`

        ${p.date}
        +$${fmt(p.amt)}
     
`).join("");
  }
}

// ─── Toast ───────────────────────────────────────────
let toastTimer;
function toast(msg){
  const el=document.getElementById("toast");
  el.textContent=msg;el.style.display="block";
  clearTimeout(toastTimer);
  toastTimer=setTimeout(()=>{el.style.display="none";},2400);
}

// ─── Clock ───────────────────────────────────────────
function updateClock(){
  const now=new Date();
  document.getElementById("clock").textContent=
    now.getHours().toString().padStart(2,"0")+":"+now.getMinutes().toString().padStart(2,"0");
}
setInterval(updateClock,10000);updateClock();
